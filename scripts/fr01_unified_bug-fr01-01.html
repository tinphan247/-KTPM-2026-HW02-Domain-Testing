// Dòng 15: Biểu thức chính quy kiểm tra mật khẩu bị lỗi ngược
const flawedStrongPasswordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*\s)[A-Za-z\d\s]{8,}$/;
 
if (!flawedStrongPasswordRegex.test(password)) {
  setError('Mật khẩu quá yếu! Phải dài tối thiểu 8 ký tự, gồm chữ hoa, chữ thường, số và KÝ TỰ ĐẶC BIỆT.');
  return;
}
⚠️ ROOT CAUSE ANALYSIS — BUG-FR01-01
Lập trình viên viết nhầm (?=.*\s) bắt buộc phải có khoảng trắng (space) thay vì ký tự đặc biệt. Đồng thời tập ký tự [A-Za-z\d\s]{8,}$ cấm toàn bộ các ký tự đặc biệt chuẩn [@$!%*?&]. Khiến mật khẩu hợp lệ bị từ chối 100%, còn mật khẩu có dấu cách lại lọt qua!
✓ RECOMMENDED FIX (PATCH)
const validStrongPasswordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$/;
