const authenticateToken = (req, res, next) => {
  const authHeader = req.headers["authorization"];
  const token = authHeader && authHeader.split(" ")[1];
  if (token == null) return res.status(401).json({ error: "Unauthorized" });
  jwt.verify(token, SECRET_KEY, (err, user) => {
    if (err) return res.status(403).json({ error: "Forbidden" });
    req.user = user; // LỖ HỔNG FATAL: Chỉ decode Token, HOÀN TOÀN KHÔNG KIỂM TRA req.user.role === 'admin'!
    next();
  });
};
// Các endpoint quản trị nhạy cảm sử dụng trực tiếp authenticateToken mà không có requireAdmin
app.post("/api/categories", authenticateToken, ...); // User thường có thể POST
app.delete("/api/categories/:id", authenticateToken, ...); // User thường có thể DELETE
⚠️ ROOT CAUSE ANALYSIS — BUG-FR14-02
Lỗ hổng Broken Access Control (OWASP Top 10 A01 / BOLA): Middleware authenticateToken chỉ kiểm tra tính hợp lệ của chữ ký JWT mà bỏ quên việc kiểm tra claim req.user.role === 'admin'. Các API quản trị danh mục dùng chung middleware này, cho phép tài khoản khách hàng thông thường (role = 'user') vượt qua phân quyền để tạo và xóa danh mục hệ thống tùy ý.
✓ RECOMMENDED FIX (PATCH)
// Tạo middleware phân quyền quản trị
const requireAdmin = (req, res, next) => {
  if (!req.user || req.user.role !== 'admin') {
    return res.status(403).json({ error: "Access denied. Admin privileges required." });
  }
  next();
};
// Áp dụng bảo vệ cho các route quản trị
app.post("/api/categories", authenticateToken, requireAdmin, ...);
app.delete("/api/categories/:id", authenticateToken, requireAdmin, ...);
